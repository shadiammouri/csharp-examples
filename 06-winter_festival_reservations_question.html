📄 Complete C# Program (Program.cs) الكود الكامل والجاهز للتشغيل 
using System;

class WinterFestival
{
    static void Main()
    {
        // Accumulators for attendees on each date
        int count10 = 0, count12 = 0, count13 = 0;

        // Loop over all 500 families
        for (int i = 1; i <= 500; i++)
        {
            Console.Write("Enter event date (10, 12, or 13): ");
            int date = int.Parse(Console.ReadLine());

            Console.Write("Enter number of people: ");
            int people = int.Parse(Console.ReadLine());

            if (date == 10)      count10 += people;
            else if (date == 12) count12 += people;
            else if (date == 13) count13 += people;
        }

        // Calculate total attendees and revenue
        int totalPeople  = count10 + count12 + count13;
        int totalRevenue = totalPeople * 30;

        // Print final festival report
        Console.WriteLine($"Attendees on December 10: {count10}");
        Console.WriteLine($"Attendees on December 12: {count12}");
        Console.WriteLine($"Attendees on December 13: {count13}");
        Console.WriteLine($"Total festival revenue: {totalRevenue} NIS");
    }
}
